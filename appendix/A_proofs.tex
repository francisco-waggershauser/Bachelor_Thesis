\begin{lemma}\label{lem:scores}
For every agent $i$ and good $g$, $\Var(\Zig)=1$. For agents $i\neq j$,
$\Corr(\Zig,Z_{jg})=\rho$.
\end{lemma}

\begin{proof}
The components $\Wg$, $\etaig$ and $\eta_{jg}$ are mutually independent, each with mean zero and
variance one. Hence
\[
  \Var(\Zig)\;=\;\rho\,\Var(\Wg)+(1-\rho)\,\Var(\etaig)\;=\;\rho+(1-\rho)\;=\;1.
\]
For $i\neq j$, all scores have mean zero, and every product of two different components has
expectation zero. Hence
\[
  \operatorname{Cov}(\Zig,Z_{jg})\;=\;\mathbb{E}\big[\Zig Z_{jg}\big]
  \;=\;\rho\,\mathbb{E}\big[\Wg^2\big]\;=\;\rho.
\]
Since both scores have variance one, their correlation equals their covariance.
\end{proof}

\begin{lemma}\label{lem:scales}
For agents $i\neq j$ and a good $g$, the Spearman rank correlation of $v_i(g)$ and $v_j(g)$ is
$\frac{6}{\pi}\arcsin(\rho/2)$. The Gini coefficient of $v_i(g)$ is $2\Phi(\theta/\sqrt{2})-1$,
where $\Phi$ denotes the standard normal distribution function.
\end{lemma}
